   133→
   134→function formatCurrency(val: number): string {
   135→  if (val >= 1_000_000) return `$${(val / 1_000_000).toFixed(1)}M`
   136→  if (val >= 1_000) return `$${(val / 1_000).toFixed(1)}K`
   137→  return `$${val.toLocaleString()}`
   138→}
