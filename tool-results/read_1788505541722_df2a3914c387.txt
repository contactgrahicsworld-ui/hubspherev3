    14→    14→function log(category, test, status, detail = '') {
    15→    15→  const icon = status === 'PASS' ? '✅' : status === 'WARN' ? '⚠️' : '❌';
    16→    16→  const msg = `${icon} [${category}] ${test}${detail ? ': ' + detail : ''}`;
    17→    17→  console.log(msg);
    18→    18→  if (status === 'PASS') results.passed++;
    19→    19→  else if (status === 'WARN') results.warnings.push(`[${category}] ${test}: ${detail}`);
    20→    20→  else { results.failed++; results.errors.push(`[${category}] ${test}: ${detail}`); }
    21→    21→}

    23→    23→async function api(path, options = {}) {
    24→    24→  const url = `${BASE}${path}`;
    25→    25→  const headers = { 'Content-Type': 'application/json', ...options.headers };
    26→    26→  if (accessToken && !options.noAuth) headers['Authorization'] = `Bearer ${accessToken}`;
    27→    27→  try {
    28→    28→    const res = await fetch(url, { ...options, headers });
    29→    29→    const text = await res.text();
    30→    30→    let data;
    31→    31→    try { data = JSON.parse(text); } catch { data = text; }
    32→    32→    return { status: res.status, data, ok: res.ok, headers: res.headers };
    33→    33→  } catch (e) {
    34→    34→    return { status: 0, data: null, ok: false, error: e.message };
    35→    35→  }
    36→    36→}

   272→   272→    // New refresh token is in Set-Cookie header (httpOnly)

   274→   274→    const hasRefreshCookie = setCookie.includes('hs-refresh-token');

   276→   276→
   277→   277→    // Old refresh token should be revoked

   279→   279→      method: 'POST',

   281→   281→      noAuth: true
   282→   282→    });
   283→   283→    log('Auth-Refresh', 'Old refresh token revoked', oldRes.status === 401 ? 'PASS' : 'FAIL', `status=${oldRes.status}`);
   284→   284→
   285→   285→    if (newAccess) accessToken = newAccess;
   286→   286→    // Extract new refresh token from cookie for subsequent tests
   287→   287→    const match = setCookie.match(/hs-refresh-token=([^;]+)/);
   288→   288→    if (match) refreshToken = match[1];
   289→   289→  }
   290→   290→
   291→   291→  res = await api('/api/v1/auth/refresh', {
   292→   292→    method: 'POST',
   293→   293→    body: JSON.stringify({ refreshToken: 'invalid-token-xyz' }),

   295→   295→  });
