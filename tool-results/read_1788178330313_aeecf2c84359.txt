   140→class TocDocTemplate(SimpleDocTemplate):
   141→    def afterFlowable(self, flowable):
   142→        if hasattr(flowable, 'bookmark_name'):
   143→            level = getattr(flowable, 'bookmark_level', 0)
   144→            text = getattr(flowable, 'bookmark_text', '')
   145→            key = getattr(flowable, 'bookmark_key', '')
   146→            self.notify('TOCEntry', (level, text, self.page, key))

   148→def add_heading(text, style, level=0):
   149→    key = f'h_{hashlib.md5(text.encode()).hexdigest()[:8]}'
   150→    p = Paragraph(f'<a name="{key}"/>{text}', style)
   151→    p.bookmark_name = key
   152→    p.bookmark_level = level
   153→    p.bookmark_text = text
   154→    p.bookmark_key = key
   155→    return p

   158→def make_table(headers, rows, col_widths=None):
   159→    cw = col_widths or [CONTENT_W / len(headers)] * len(headers)
   160→    hdr = [Paragraph(h, styles['TableHead']) for h in headers]
   161→    data = [hdr]
   162→    for row in rows:
   163→        data.append([Paragraph(str(c), styles['TableCell']) for c in row])
   164→    t = Table(data, colWidths=cw, repeatRows=1)
   165→    style_cmds = [
   166→        ('BACKGROUND', (0, 0), (-1, 0), HEADER_FILL),
   167→        ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
   168→        ('ALIGN', (0, 0), (-1, -1), 'LEFT'),
   169→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),

   171→        ('TOPPADDING', (0, 0), (-1, -1), 5),
   172→        ('BOTTOMPADDING', (0, 0), (-1, -1), 5),
   173→        ('LEFTPADDING', (0, 0), (-1, -1), 6),
   174→        ('RIGHTPADDING', (0, 0), (-1, -1), 6),
   175→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
