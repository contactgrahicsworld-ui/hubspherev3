   159→function formatDate(dateStr: string) {
   160→  if (!dateStr) return '-'
   161→  try {
   162→    return new Date(dateStr).toLocaleDateString('en-US', {
   163→      month: 'short',
   164→      day: 'numeric',
   165→      year: 'numeric',
   166→    })
   167→  } catch {
   168→    return '-'
   169→  }
   170→}
   171→
   172→function formatDateTime(dateStr: string) {
   173→  if (!dateStr) return '-'
   174→  try {
   175→    return new Date(dateStr).toLocaleDateString('en-US', {
   176→      month: 'short',
   177→      day: 'numeric',
   178→      year: 'numeric',
   179→      hour: '2-digit',
   180→      minute: '2-digit',
   181→    })
   182→  } catch {
   183→    return '-'
   184→  }
   185→}
   186→
   187→function formatCurrency(val: number | null): string {
   188→  if (val == null) return '-'
   189→  return new Intl.NumberFormat('en-US', {
   190→    style: 'currency',
   191→    currency: 'USD',
   192→    minimumFractionDigits: 0,
   193→    maximumFractionDigits: 0,
   194→  }).format(val)
   195→}
   196→
   197→function getLeadName(lead: Lead): string {
   198→  return [lead.firstName, lead.lastName].filter(Boolean).join(' ') || '-'
   199→}
   200→
   201→function timeAgo(dateStr: string): string {
   202→  const now = new Date()
   203→  const date = new Date(dateStr)
   204→  const seconds = Math.floor((now.getTime() - date.getTime()) / 1000)
   205→  if (seconds < 60) return 'just now'
   206→  if (seconds < 3600) return `${Math.floor(seconds / 60)}m ago`
   207→  if (seconds < 86400) return `${Math.floor(seconds / 3600)}h ago`
   208→  if (seconds < 604800) return `${Math.floor(seconds / 86400)}d ago`
   209→  return formatDate(dateStr)
   210→}
