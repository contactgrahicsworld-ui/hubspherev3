    10→    10→function log(m: string, t: string, ok: boolean, d?: string) {
    11→    11→  if (!RES[m]) RES[m] = { p: 0, f: 0, e: [] };
    12→    12→  if (ok) RES[m].p++;
    13→    13→  else {
    14→    14→    RES[m].f++;
    15→    15→    RES[m].e.push(t + ': ' + (d || ''));
    16→    16→  }
    17→    17→}

    27→    27→    });
    28→    28→    const d = await r.json().catch(() => null);

    30→    30→  } catch (e) {
    31→    31→    return { s: 0, d: { error: String(e) }, o: false };
    32→    32→  }
