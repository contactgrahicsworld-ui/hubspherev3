    71→class Output:
    72→    """Structured JSON output for all subcommands."""

    74→    @staticmethod
    75→    def success(data: dict):
    76→        payload = {"status": "success", "data": data}
    77→        sys.stdout.write(json.dumps(payload, ensure_ascii=False, indent=2) + "\n")
    78→        raise SystemExit(0)
    79→
    80→    @staticmethod
    81→    def error(error: str, message: str, hint: Optional[str] = None, code: int = 1):
    82→        payload = {"status": "error", "error": error, "message": message}
    83→        if hint is not None:
    84→            payload["hint"] = hint
    85→        sys.stderr.write(json.dumps(payload, ensure_ascii=False, indent=2) + "\n")
    86→        raise SystemExit(code)

    88→    @staticmethod
    89→    def check_file(filepath: str) -> Path:
    90→        target = Path(filepath)
    91→        if not target.exists():
    92→            Output.error("FileNotFound", f"File not found: {filepath}", code=2)
    93→        return target

   100→def cmd(name: str):
   101→    """Decorator to register a CLI command under a dotted namespace."""
   102→    def decorator(fn: Callable) -> Callable:
   103→        _COMMANDS[name] = fn

   108→def _pop_flag(argv: list, short: str, long: str, needs_value: bool = True):
   109→    """Extract a flag (and optional value) from *argv* in-place."""
   110→    for idx, tok in enumerate(argv):
   111→        if tok in (short, long):
   112→            argv.pop(idx)
   113→            if needs_value:
   114→                if idx < len(argv):
   115→                    return argv.pop(idx)
   116→                Output.error("MissingArg", f"Flag {long} requires a value")
   117→            return True
   118→    return None
   119→

   121→def _load_json_arg(argv: list) -> dict:
   122→    """Read JSON from -d/--data string or -f/--file path."""
   123→    raw = _pop_flag(argv, "-d", "--data")
   124→    if raw is not None:
   125→        try:
   126→            return json.loads(raw)
   127→        except json.JSONDecodeError as exc:
   128→            Output.error("InvalidJSON", f"JSON parse error: {exc}")
   129→
   130→    fpath = _pop_flag(argv, "-f", "--file")
   131→    if fpath is not None:
   132→        try:
   133→            with open(fpath) as fh:
   134→                return json.load(fh)
   135→        except Exception as exc:
   136→            Output.error("FileError", f"Failed to read file: {exc}")
   137→
   138→    Output.error("MissingData", "Requires --data or --file argument")

   141→def _resolve_page_indices(range_spec: Optional[str], page_count: int) -> List[int]:
   142→    """Turn a human-friendly range string (1-indexed) into a sorted list of 0-based indices."""
   143→    if not range_spec:
   144→        return list(range(page_count))
   145→    indices: Set[int] = set()
   146→    for segment in range_spec.split(","):
   147→        segment = segment.strip()
   148→        if "-" in segment:
   149→            lo, hi = segment.split("-", 1)
   150→            for i in range(int(lo) - 1, min(int(hi), page_count)):
   151→                indices.add(i)
   152→        else:
   153→            val = int(segment) - 1
   154→            if 0 <= val < page_count:
   155→                indices.add(val)
   156→    return sorted(indices)

   166→def _probe_cmd(name: str, version_args: Optional[List[str]] = None) -> Tuple[str, str]:
   167→    """Check if a command exists and optionally get its version. Returns (status, detail)."""
   168→    path = shutil.which(name)
   169→    if path is None:
   170→        return ("missing", "")
   171→    if version_args is None:
   172→        return ("ok", "")
   173→    try:
   174→        result = subprocess.run(
   175→            [path] + version_args,
   176→            capture_output=True, text=True, timeout=10
   177→        )
   178→        ver = result.stdout.strip() or result.stderr.strip()
   179→        return ("ok", ver)
   180→    except Exception:
   181→        return ("ok", "")

   184→def _probe_python_module(mod_name: str) -> Tuple[str, str]:
   185→    """Check if a Python module is importable and get its version."""
   186→    try:
   187→        result = subprocess.run(
   188→            [sys.executable, "-c", f"import {mod_name}; print(getattr({mod_name}, '__version__', 'installed'))"],
   189→            capture_output=True, text=True, timeout=10
   190→        )
   191→        if result.returncode == 0:
   192→            return ("ok", result.stdout.strip())
   193→        return ("missing", "")
   194→    except Exception:
   195→        return ("missing", "")

   244→def _probe_playwright_npm() -> Tuple[str, str]:
   245→    """Check if playwright npm package is installed."""
   246→    try:
   247→        result = subprocess.run(
   248→            ["node", "-e", "console.log(require('playwright/package.json').version)"],
   249→            capture_output=True, text=True, timeout=10
   250→        )
   251→        if result.returncode == 0 and result.stdout.strip():
   252→            return ("ok", result.stdout.strip())
   253→    except Exception:
   254→        pass
   255→    # Try global
   256→    try:
   257→        result = subprocess.run(
   258→            ["npm", "list", "-g", "playwright", "--depth=0"],
   259→            capture_output=True, text=True, timeout=15
   260→        )
   261→        import re as _re
   262→        m = _re.search(r"playwright@(\S+)", result.stdout)
   263→        if m:
   264→            return ("ok", m.group(1))
   265→    except Exception:
   266→        pass

   369→@cmd("env.fix")
   370→def env_fix(argv: list):
   371→    """Auto-install missing Python dependencies."""
   372→    modules = {
   373→        "pikepdf": "pikepdf",
   374→        "pdfplumber": "pdfplumber",
   375→        "pypdf": "pypdf",
   376→        "pdf2image": "pdf2image",
   377→        "PIL": "Pillow",
   378→    }
   379→    installed = []
   380→    for mod, pkg in modules.items():
   381→        s, _ = _probe_python_module(mod)
   382→        if s == "missing":
   383→            print(f"Installing {pkg}...")
   384→            for attempt in (
   385→                [sys.executable, "-m", "pip", "install", "-q", pkg],
   386→                [sys.executable, "-m", "pip", "install", "-q", "--user", pkg],
   387→                [sys.executable, "-m", "pip", "install", "-q", "--break-system-packages", pkg],
   388→            ):
   389→                result = subprocess.run(attempt, capture_output=True, text=True)
   390→                if result.returncode == 0:
   391→                    installed.append(pkg)
   392→                    break
   393→            else:
   394→                print(f"  Failed to install {pkg}")
   395→
   396→    if installed:
   397→        print(f"\nInstalled: {', '.join(installed)}")
   398→    else:
   399→        print("All Python dependencies are already installed.")
   400→    raise SystemExit(0)

   407→@cmd("extract.text")
   408→def extract_text(argv: list):
   409→    """Pull plain text from selected pages."""
   410→    if not argv:
   411→        Output.error("MissingArg", "pdf path required")
   412→    pdf_path = argv.pop(0)
   413→    page_range = _pop_flag(argv, "-p", "--pages")
   414→
   415→    import pdfplumber
   416→    src = Output.check_file(pdf_path)
   417→    try:
   418→        doc = pdfplumber.open(src)
   419→    except Exception as exc:
   420→        Output.error("PDFError", f"Cannot open PDF: {exc}", code=3)
   421→
   422→    total_pages = len(doc.pages)
   423→    target_pages = _resolve_page_indices(page_range, total_pages)
   424→    char_total = 0
   425→    page_results = []
   426→
   427→    for pg_idx in target_pages:
   428→        content = doc.pages[pg_idx].extract_text() or ""
   429→        char_total += len(content)
   430→        page_results.append({"page": pg_idx + 1, "chars": len(content), "text": content})
   431→
   432→    doc.close()
   433→    Output.success({
   434→        "total_pages": total_pages,
   435→        "extracted_pages": len(target_pages),
   436→        "total_chars": char_total,
   437→        "pages": page_results,
   438→    })

   441→@cmd("extract.table")
   442→def extract_table(argv: list):
   443→    """Locate and return every table on selected pages."""
   444→    if not argv:
   445→        Output.error("MissingArg", "pdf path required")
   446→    pdf_path = argv.pop(0)
   447→    page_range = _pop_flag(argv, "-p", "--pages")
   448→
   449→    import pdfplumber
   450→    src = Output.check_file(pdf_path)
   451→    try:
   452→        doc = pdfplumber.open(src)
   453→    except Exception as exc:
   454→        Output.error("PDFError", f"Cannot open PDF: {exc}", code=3)
   455→
   456→    total_pages = len(doc.pages)
   457→    target_pages = _resolve_page_indices(page_range, total_pages)
   458→    collected = []
   459→
   460→    for pg_idx in target_pages:
   461→        for tbl_num, raw_table in enumerate(doc.pages[pg_idx].extract_tables()):
   462→            if not raw_table:
   463→                continue
   464→            sanitised = [
   465→                [(cell.strip() if cell else "") for cell in row]
   466→                for row in raw_table
   467→            ]
   468→            collected.append({
   469→                "page": pg_idx + 1,
   470→                "table_index": tbl_num,
   471→                "rows": len(sanitised),
   472→                "cols": len(sanitised[0]) if sanitised else 0,
   473→                "data": sanitised,
   474→            })
   475→
   476→    doc.close()
   477→    Output.success({
   478→        "total_pages": total_pages,
   479→        "extracted_pages": len(target_pages),
   480→        "total_tables": len(collected),
   481→        "tables": collected,
   482→    })

   485→@cmd("extract.image")
   486→def extract_image(argv: list):
   487→    """Save every embedded raster image to output dir."""
   488→    if not argv:
   489→        Output.error("MissingArg", "pdf path required")
   490→    pdf_path = argv.pop(0)
   491→    out_dir = _pop_flag(argv, "-o", "--output") or "."
   492→
   493→    import pikepdf
   494→    src = Output.check_file(pdf_path)
   495→    dest = Path(out_dir)
   496→    dest.mkdir(parents=True, exist_ok=True)
   497→
   498→    try:
   499→        doc = pikepdf.open(src)
   500→    except Exception as exc:
   501→        Output.error("PDFError", f"Cannot open PDF: {exc}", code=3)
   502→
   503→    saved = []
   504→    seq = 0
   505→    _EXT_MAP = {
   506→        "/DCTDecode": "jpg",
   507→        "/FlateDecode": "png",
   508→        "/JPXDecode": "jp2",
   509→    }
   510→
   511→    for page_no, pg in enumerate(doc.pages, 1):
   512→        res = pg.get("/Resources")
   513→        if res is None or "/XObject" not in res:
   514→            continue
   515→        for key, ref in res.XObject.items():
   516→            try:
   517→                img_obj = doc.get_object(ref.objgen)
   518→                if img_obj.get("/Subtype") != "/Image":
   519→                    continue
   520→                seq += 1
   521→                w = int(img_obj.get("/Width", 0))
   522→                h = int(img_obj.get("/Height", 0))
   523→                filt = img_obj.get("/Filter")
   524→                ext = _EXT_MAP.get(str(filt) if filt else None, "bin")
   525→                fname = f"page{page_no}_img{seq}.{ext}"
   526→                out_file = dest / fname
   527→                out_file.write_bytes(img_obj.read_raw_bytes())
   528→                saved.append({
   529→                    "page": page_no, "name": str(key), "file": str(out_file),
   530→                    "width": w, "height": h, "format": ext,
   531→                })
   532→            except Exception:
   533→                continue
   534→
   535→    doc.close()
   536→    Output.success({"output_dir": str(dest), "total_images": len(saved), "images": saved})

   543→@cmd("pages.merge")
   544→def pages_merge(argv: list):
   545→    """Concatenate several PDF files into one."""
   546→    out_path = _pop_flag(argv, "-o", "--output")
   547→    if out_path is None:
   548→        Output.error("MissingArg", "--output is required")
   549→    if not argv:
   550→        Output.error("MissingArg", "At least one PDF required")
   551→
   552→    import pikepdf
   553→    sources = [Output.check_file(p) for p in argv]
   554→    handles = []
   555→    try:
   556→        combined = pikepdf.new()
   557→        descriptions = []
   558→        for src in sources:
   559→            handle = pikepdf.open(src)
   560→            handles.append(handle)
   561→            n = len(handle.pages)
   562→            descriptions.append(f"{src.name} ({n} pages)")
   563→            for pg in handle.pages:
   564→                combined.pages.append(pg)
   565→        total = len(combined.pages)
   566→        combined.save(out_path)
   567→        combined.close()
   568→    except Exception as exc:

   695→@cmd("pages.clean")
   696→def pages_clean(argv: list):
   697→    """Remove truly blank pages from a PDF.
   698→
   699→    A page is considered blank ONLY if it has exactly 0 text characters AND
   700→    0 images.  Pages with even a single character or a tiny image are kept.
   701→
   702→    Usage:
   703→        pdf.py pages.clean input.pdf -o output.pdf
   704→    """
   705→    if not argv:
   706→        Output.error("MissingArg", "PDF path required")
   707→    pdf_path = argv.pop(0)
   708→    out_path = _pop_flag(argv, "-o", "--output")
   709→    if out_path is None:
   710→        Output.error("MissingArg", "--output is required")
   711→    # --threshold is accepted but ignored (kept for backward compat)
   712→    _pop_flag(argv, "-t", "--threshold")
   713→
   714→    src = Output.check_file(pdf_path)
   715→
   716→    # Phase 1: Detect blank pages using pdfplumber (text extraction)
   717→    try:
   718→        import pdfplumber
   719→    except ImportError:
   720→        Output.error("DependencyMissing", "pdfplumber required: pip install pdfplumber")
   721→
   722→    blank_indices = []  # 0-indexed
   723→    total_pages = 0
   724→    try:
   725→        pdf = pdfplumber.open(str(src))
   726→        total_pages = len(pdf.pages)
   727→        for i, page in enumerate(pdf.pages):
   728→            text = (page.extract_text() or "").strip()
   729→            # Check for ANY images on the page (no size filter)
   730→            images = page.images if hasattr(page, 'images') else []
   731→            # A page is blank ONLY if it has exactly 0 characters AND 0 images
   732→            if len(text) == 0 and len(images) == 0:
   733→                blank_indices.append(i)
   734→        pdf.close()
   735→    except Exception as exc:
   736→        Output.error("PDFError", f"Cannot analyze PDF: {exc}", code=3)
   737→
   738→    if not blank_indices:
   739→        Output.success({
   740→            "output": str(src),
   741→            "total_pages": total_pages,
   742→            "blank_pages_removed": 0,
   743→            "message": "No blank pages found",
   744→        })
   745→        return
   746→
   747→    # Phase 2: Remove blank pages using pikepdf
   748→    try:
   749→        import pikepdf
   750→    except ImportError:
   751→        Output.error("DependencyMissing", "pikepdf required: pip install pikepdf")
   752→
   753→    try:
   754→        doc = pikepdf.open(str(src))
   755→        # Remove in reverse order to preserve indices
   756→        for idx in sorted(blank_indices, reverse=True):
   757→            if idx < len(doc.pages):
   758→                del doc.pages[idx]
   759→        doc.save(out_path)
   760→        doc.close()
   761→    except Exception as exc:
   762→        Output.error("PDFError", f"Cannot remove blank pages: {exc}", code=4)
   763→
   764→    remaining = total_pages - len(blank_indices)
   765→    Output.success({
   766→        "output": out_path,
   767→        "total_pages": total_pages,
   768→        "blank_pages_removed": len(blank_indices),
   769→        "blank_page_numbers": [i + 1 for i in blank_indices],  # 1-indexed for humans
   770→        "remaining_pages": remaining,
   771→    })

   956→
   957→_FIELD_TYPE_MAP = {
   958→    "/Tx": "text",
   959→    "/Sig": "signature",
   960→}
   961→
   962→
   963→def _classify_field(node) -> str:
   964→    """Map a PDF field type token to a human label."""
   965→    ft = str(node.get("/FT", ""))
   966→    if ft in _FIELD_TYPE_MAP:
   967→        return _FIELD_TYPE_MAP[ft]
   968→    flags = int(node.get("/Ff", 0))
   969→    if ft == "/Btn":
   970→        return "radio" if (flags & (1 << 15)) else "checkbox"
   971→    if ft == "/Ch":
   972→        return "dropdown" if (flags & (1 << 17)) else "listbox"
   973→    return "unknown"
   974→
   975→
   976→def _extra_props(node, kind: str) -> dict:
   977→    """Gather type-specific metadata (options, checked value, etc.)."""
   978→    props: dict = {}
   979→    if kind == "checkbox":
   980→        ap = node.get("/AP")
   981→        if ap and "/N" in ap:
   982→            states = [str(s) for s in ap["/N"].keys()]
   983→            props["states"] = states
   984→            props["checked_value"] = next((s for s in states if s != "/Off"), states[0] if states else None)
   985→    elif kind in ("dropdown", "listbox"):
   986→        raw_opts = node.get("/Opt")
   987→        if raw_opts:
   988→            props["options"] = [
   989→                {"value": str(item[0]), "label": str(item[1])} if isinstance(item, list) and len(item) >= 2
   990→                else {"value": str(item), "label": str(item)}
   991→                for item in raw_opts
   992→            ]
   993→    elif kind == "radio":
   994→        kids = node.get("/Kids")
   995→        if kids:
   996→            radio_vals = []
   997→            for child in kids:
   998→                ap = child.get("/AP")
   999→                if ap and "/N" in ap:
  1000→                    radio_vals.extend(str(k) for k in ap["/N"].keys() if str(k) != "/Off")
  1001→            if radio_vals:
  1002→                props["options"] = radio_vals
  1003→    return props
  1004→
  1005→
  1006→def _current_value(node):
  1007→    v = node.get("/V")
  1008→    return str(v) if v is not None else None
  1009→
  1010→
  1011→def _gather_fields(doc) -> list:
  1012→    """Walk the AcroForm field tree iteratively and return a flat list."""
  1013→    if "/AcroForm" not in doc.Root:
  1014→        return []
  1015→    acro = doc.Root.AcroForm
  1016→    if "/Fields" not in acro:
  1017→        return []
  1018→
  1019→    page_lookup = {pg.objgen: idx for idx, pg in enumerate(doc.pages)}
  1020→    results = []
  1021→    stack = [(field, "") for field in reversed(list(acro.Fields))]
  1022→
  1023→    while stack:
  1024→        node, parent_path = stack.pop()
  1025→        name = str(node.get("/T", ""))
  1026→        full = f"{parent_path}.{name}" if parent_path else name
  1027→
  1028→        kids = node.get("/Kids")
  1029→        if kids and any("/T" in k for k in kids):
  1030→            for kid in reversed(list(kids)):
  1031→                stack.append((kid, full))
  1032→            continue
  1033→
  1034→        kind = _classify_field(node)
  1035→        if kind == "unknown":
  1036→            continue
  1037→
  1038→        entry = {"id": full, "type": kind}
  1039→        val = _current_value(node)
  1040→        if val:
  1041→            entry["current_value"] = val
  1042→        entry.update(_extra_props(node, kind))
  1043→
  1044→        page_ref = node.get("/P")
  1045→        if page_ref and hasattr(page_ref, "objgen"):
  1046→            pg_num = page_lookup.get(page_ref.objgen)
  1047→            if pg_num is not None:
  1048→                entry["page"] = pg_num + 1
  1049→
  1050→        results.append(entry)
  1051→
  1052→    return results

  1074→@cmd("form.fill")
  1075→def form_fill(argv: list):
  1076→    """Write values into a fillable PDF (pikepdf version)."""
  1077→    if not argv:
  1078→        Output.error("MissingArg", "pdf path required")
  1079→    pdf_path = argv.pop(0)
  1080→    out_path = _pop_flag(argv, "-o", "--output")
  1081→    if out_path is None:
  1082→        Output.error("MissingArg", "--output is required")
  1083→    data = _load_json_arg(argv)
  1084→
  1085→    import pikepdf
  1086→    src = Output.check_file(pdf_path)
  1087→    try:
  1088→        doc = pikepdf.open(src)
  1089→    except Exception as exc:
  1090→        Output.error("PDFError", f"Cannot open PDF: {exc}", code=3)
  1091→
  1092→    if "/AcroForm" not in doc.Root or "/Fields" not in doc.Root.AcroForm:
  1093→        Output.error("NoForm", "This PDF has no form fields")
  1094→
  1095→    known = {f["id"]: f for f in _gather_fields(doc)}
  1096→
  1097→    # Validation
  1098→    issues = []
  1099→    for fid, fval in data.items():
  1100→        if fid not in known:
  1101→            issues.append(f"Field not found: {fid}")
  1102→            continue
  1103→        fmeta = known[fid]
  1104→        ftype = fmeta["type"]
  1105→        if ftype == "checkbox" and "states" in fmeta:
  1106→            ok_vals = fmeta["states"]
  1107→            if fval not in ok_vals and f"/{fval}" not in ok_vals and fval not in ("true", "True", "false", "False", "1", "0"):
  1108→                issues.append(f"Invalid value for field {fid}, options: {ok_vals} or true/false")
  1109→        if ftype in ("dropdown", "listbox") and "options" in fmeta:
  1110→            ok_vals = [o["value"] for o in fmeta["options"]]
  1111→            if fval not in ok_vals:
  1112→                issues.append(f"Invalid value for field {fid}, options: {ok_vals}")
  1113→    if issues:
  1114→        Output.error("ValidationError", "Field validation failed", hint="; ".join(issues))
  1115→
  1116→    # Fill
  1117→    written = 0
  1118→
  1119→    def _apply(node, parent_path=""):
  1120→        nonlocal written
  1121→        name = str(node.get("/T", ""))
  1122→        full = f"{parent_path}.{name}" if parent_path else name
  1123→        kids = node.get("/Kids")
  1124→        if kids and any("/T" in k for k in kids):
  1125→            for kid in kids:
  1126→                _apply(kid, full)
  1127→            return
  1128→        if full not in data:

  1204→def _get_field_info(reader) -> list:
  1205→    """Extract detailed field info from a PdfReader, including radio group aggregation."""
  1206→    fields = reader.get_fields()
  1207→
  1208→    field_info_by_id = {}

  1210→

  1212→        if field.get("/Kids"):
  1213→            if field.get("/FT") == "/Btn":
  1214→                possible_radio_names.add(field_id)
  1215→            continue

  1217→
  1218→    radio_fields_by_id: Dict[str, dict] = {}
  1219→

  1221→        annotations = page.get('/Annots', [])
  1222→        for ann in annotations:
  1223→            field_id = _get_full_annotation_field_id(ann)
  1224→            if field_id in field_info_by_id:
  1225→                field_info_by_id[field_id]["page"] = page_index + 1
  1226→                field_info_by_id[field_id]["rect"] = ann.get('/Rect')
  1227→            elif field_id in possible_radio_names:
  1228→                try:
  1229→                    on_values = [v for v in ann["/AP"]["/N"] if v != "/Off"]
  1230→                except KeyError:
  1231→                    continue
  1232→                if len(on_values) == 1:
  1233→                    rect = ann.get("/Rect")
  1234→                    if field_id not in radio_fields_by_id:
  1235→                        radio_fields_by_id[field_id] = {
  1236→                            "field_id": field_id,
  1237→                            "type": "radio_group",
  1238→                            "page": page_index + 1,
  1239→                            "radio_options": [],
  1240→                        }
  1241→                    radio_fields_by_id[field_id]["radio_options"].append({
  1242→                        "value": on_values[0],
  1243→                        "rect": rect,
  1244→                    })

  1247→    fields_with_location = []
  1248→    for field_info in field_info_by_id.values():
  1249→        if "page" in field_info:
  1250→            fields_with_location.append(field_info)
  1251→        else:
  1252→            print(f"Unable to determine location for field id: {field_info.get('field_id')}, ignoring")
  1253→
  1254→    # Sort by page number, then Y position (flipped), then X
  1255→    def sort_key(f):
  1256→        if "radio_options" in f:
  1257→            rect = f["radio_options"][0]["rect"] or [0, 0, 0, 0]
  1258→        else:
  1259→            rect = f.get("rect") or [0, 0, 0, 0]
  1260→        adjusted_position = [-rect[1], rect[0]]
  1261→        return [f.get("page"), adjusted_position]
  1262→
  1263→    sorted_fields = fields_with_location + list(radio_fields_by_id.values())

  1307→def _monkeypatch_pypdf_method():
  1308→    """
  1309→    Workaround for pypdf bug with selection list fields.
  1310→    pypdf's get_inherited returns a list of two-element lists for /Opt fields
  1311→    in selection lists, causing join() to throw TypeError. We patch it to
  1312→    return just the value strings.
  1313→    """

  1315→    from pypdf.constants import FieldDictionaryAttributes
  1316→
  1317→    original_get_inherited = DictionaryObject.get_inherited
  1318→
  1319→    def patched_get_inherited(self, key: str, default=None):
  1320→        result = original_get_inherited(self, key, default)
  1321→        if key == FieldDictionaryAttributes.Opt:
  1322→            if isinstance(result, list) and all(isinstance(v, list) and len(v) == 2 for v in result):
  1323→                result = [r[0] for r in result]
  1324→        return result
  1325→
  1326→    DictionaryObject.get_inherited = patched_get_inherited

  1392→def _transform_coordinates(bbox, image_width, image_height, pdf_width, pdf_height):
  1393→    """Transform bounding box from image coordinates to PDF coordinates."""
  1394→    x_scale = pdf_width / image_width
  1395→    y_scale = pdf_height / image_height
  1396→
  1397→    left = bbox[0] * x_scale
  1398→    right = bbox[2] * x_scale
  1399→
  1400→    # Flip Y coordinates for PDF
  1401→    top = pdf_height - (bbox[1] * y_scale)
  1402→    bottom = pdf_height - (bbox[3] * y_scale)
  1403→
  1404→    return left, bottom, right, top

  1407→def _normalise_fields_json(raw: dict) -> dict:
  1408→    """Accept both the current sheet-based schema and the legacy flat schema.
  1409→
  1410→    Current (v2) schema uses ``sheet[].pg/dims/regions[]`` with nested
  1411→    ``label.bbox``, ``target.bbox``, ``ink{}``.
  1412→
  1413→    Legacy (v1) schema uses ``pages[]`` + ``form_fields[]`` with flat keys
  1414→    like ``entry_bounding_box``, ``label_bounding_box``, ``entry_text{}``.
  1415→
  1416→    Returns a normalised dict in the **v2** internal format used by all
  1417→    downstream functions.
  1418→    """
  1419→    # Already v2
  1420→    if "sheet" in raw:
  1421→        return raw
  1422→
  1423→    # Convert legacy → v2
  1424→    pages_lut = {p["page_number"]: p for p in raw.get("pages", [])}
  1425→    sheets: dict = {}  # pg -> sheet entry
  1426→
  1427→    for f in raw.get("form_fields", []):
  1428→        pg = f["page_number"]
  1429→        if pg not in sheets:
  1430→            pi = pages_lut.get(pg, {})
  1431→            sheets[pg] = {
  1432→                "pg": pg,
  1433→                "dims": [pi.get("image_width", 0), pi.get("image_height", 0)],
  1434→                "regions": [],
  1435→            }
  1436→        et = f.get("entry_text", {})
  1437→        region = {
  1438→            "id": f.get("field_label", f.get("description", "")),
  1439→            "hint": f.get("description", ""),
  1440→            "label": {"tag": f.get("field_label", ""), "bbox": f.get("label_bounding_box", [0, 0, 0, 0])},
  1441→            "target": {"bbox": f.get("entry_bounding_box", [0, 0, 0, 0])},
  1442→            "ink": {},
  1443→        }
  1444→        if isinstance(et, dict) and et.get("text"):
  1445→            region["ink"]["value"] = et["text"]
  1446→            if "font_size" in et:
  1447→                region["ink"]["size"] = et["font_size"]
  1448→            if "font_color" in et:
  1449→                region["ink"]["color"] = et["font_color"]
  1450→            if "font" in et:
  1451→                region["ink"]["font"] = et["font"]
  1452→        sheets[pg]["regions"].append(region)
  1453→
  1454→    return {"sheet": list(sheets.values())}

  1593→# --- form.check-bbox (bounding box overlap detection) ---
  1594→
  1595→@dataclass
  1596→class _RectAndField:
  1597→    rect: list
  1598→    rect_type: str
  1599→    field: dict
  1600→
  1601→
  1602→def get_bounding_box_messages(fields_json_stream) -> List[str]:
  1603→    """Check for overlapping bounding boxes. Returns list of messages (max 20)."""
  1604→    messages = []
  1605→    raw = json.load(fields_json_stream)
  1606→    data = _normalise_fields_json(raw)
  1607→
  1608→    total_regions = sum(len(pe["regions"]) for pe in data["sheet"])
  1609→    messages.append(f"Read {total_regions} regions across {len(data['sheet'])} page(s)")
  1610→
  1611→    def rects_intersect(r1, r2):
  1612→        disjoint_horizontal = r1[0] >= r2[2] or r1[2] <= r2[0]
  1613→        disjoint_vertical = r1[1] >= r2[3] or r1[3] <= r2[1]
  1614→        return not (disjoint_horizontal or disjoint_vertical)
  1615→
  1616→    has_error = False
  1617→
  1618→    for page_entry in data["sheet"]:
  1619→        pg = page_entry["pg"]
  1620→        # Collect all rects on this page
  1621→        rects_and_regions = []
  1622→        for region in page_entry["regions"]:
  1623→            rects_and_regions.append(_RectAndField(region["label"]["bbox"], "label", region))
  1624→            rects_and_regions.append(_RectAndField(region["target"]["bbox"], "target", region))
  1625→
  1626→        for i, ri in enumerate(rects_and_regions):
  1627→            for j in range(i + 1, len(rects_and_regions)):
  1628→                rj = rects_and_regions[j]
