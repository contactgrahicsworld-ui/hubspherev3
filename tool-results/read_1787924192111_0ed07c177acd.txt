   160→function formatDate(dateStr: string) {
   161→  if (!dateStr) return '-'
   162→  try {
   163→    return new Date(dateStr).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })
   164→  } catch { return '-' }
   165→}
   166→
   167→function formatCurrency(val: number, currency?: string | null): string {
   168→  if (val == null) return '-'
   169→  return new Intl.NumberFormat('en-US', {
   170→    style: 'currency', currency: currency || 'INR', minimumFractionDigits: 0, maximumFractionDigits: 0,
   171→  }).format(val)
   172→}
   173→
   174→function stageBadge(stage: string) {
   175→  return <Badge variant='outline' className={STAGE_BADGE_STYLES[stage] || ''}>{stage}</Badge>
   176→}

   304→export default function DealsListPage() {

   307→  const [deals, setDeals] = useState<Deal[]>([])

   313→
   314→  const [search, setSearch] = useState('')
   315→  const [debouncedSearch, setDebouncedSearch] = useState('')
   316→  const [stage, setStage] = useState('')

   319→  const [ownerId, setOwnerId] = useState('')
   320→  const [sort, setSort] = useState('createdAt-desc')
   321→
   322→  const [viewMode, setViewMode] = useState<'table' | 'kanban'>('table')
   323→  const [formOpen, setFormOpen] = useState(false)
   324→
   325→  // Filter options
   326→  const [users, setUsers] = useState<UserOption[]>([])
   327→  const [contacts, setContacts] = useState<ContactOption[]>([])
   328→  const [companies, setCompanies] = useState<CompanyOption[]>([])
   329→
   330→  const debounceRef = useRef<ReturnType<typeof setTimeout>>()
   331→  useEffect(() => {
   332→    debounceRef.current = setTimeout(() => { setDebouncedSearch(search) }, 300)
   333→    return () => { if (debounceRef.current) clearTimeout(debounceRef.current) }
   334→  }, [search])
   335→
   336→  // Fetch filter options
   337→  useEffect(() => {
   338→    const fetchOptions = async () => {
   339→      try {
   340→        const [usersRes, contactsRes, companiesRes] = await Promise.all([
   341→          apiFetch<{ success: boolean; data: UserOption[] }>('/api/v1/admin/users?limit=100'),
   342→          apiFetch<{ success: boolean; data: ContactOption[] }>('/api/v1/crm/contacts?limit=100'),
   343→          apiFetch<{ success: boolean; data: CompanyOption[] }>('/api/v1/crm/companies?limit=100'),
   344→        ])
   345→        setUsers(usersRes.data ?? [])
   346→        setContacts(contactsRes.data ?? [])
   347→        setCompanies(companiesRes.data ?? [])
   348→      } catch { /* silent */ }
   349→    }
   350→    fetchOptions()
   351→  }, [])
   352→
   353→  // For Kanban view, fetch all deals (limit=200)
   354→  const [allDeals, setAllDeals] = useState<Deal[]>([])
   355→  const [loadingKanban, setLoadingKanban] = useState(false)
   356→
   357→  const fetchDeals = useCallback(async () => {
   358→    try {
   359→      setLoading(true)
   360→      setError(null)
   361→
   362→      const [sortBy, sortOrder] = sort.split('-')
   363→      const params = new URLSearchParams({ page: String(page), limit: '20', sortBy, sortOrder })
   364→
   365→      if (debouncedSearch) params.set('search', debouncedSearch)
   366→      if (stage) params.set('stage', stage)
   367→      if (contactId) params.set('contactId', contactId)
   368→      if (companyId) params.set('companyId', companyId)
   369→      if (ownerId) params.set('ownerId', ownerId)
   370→
   371→      const data = await apiFetch<PaginatedResponse>(`/api/v1/crm/deals?${params}`)
   372→      setDeals(data.data)
   373→      setTotalPages(data.pagination.totalPages)
   374→      setTotal(data.pagination.total)
   375→    } catch (err) {
   376→      const message = err instanceof Error ? err.message : 'Failed to load deals'
   377→      setError(message)
   378→      toast.error(message)
   379→    } finally {
   380→      setLoading(false)
   381→    }
   382→  }, [page, debouncedSearch, stage, contactId, companyId, ownerId, sort])
   383→
   384→  const fetchAllDeals = useCallback(async () => {
   385→    setLoadingKanban(true)
   386→    try {
   387→      const params = new URLSearchParams({ limit: '200' })
   388→      if (debouncedSearch) params.set('search', debouncedSearch)
   389→      if (contactId) params.set('contactId', contactId)
   390→      if (companyId) params.set('companyId', companyId)
   391→      if (ownerId) params.set('ownerId', ownerId)
   392→      const data = await apiFetch<PaginatedResponse>(`/api/v1/crm/deals?${params}`)
   393→      setAllDeals(data.data)
   394→    } catch { /* silent */ } finally {
   395→      setLoadingKanban(false)
   396→    }
   397→  }, [debouncedSearch, contactId, companyId, ownerId])
   398→
   399→  useEffect(() => { fetchDeals() }, [fetchDeals])
   400→  useEffect(() => {
   401→    if (viewMode === 'kanban') fetchAllDeals()
   402→  }, [viewMode, fetchAllDeals])
   403→
   404→  useEffect(() => { setPage(1) }, [debouncedSearch, stage, contactId, companyId, ownerId, sort])
   405→
   406→  const renderPageNumbers = () => {
   407→    const pages: number[] = []
   408→    const maxVisible = 5
   409→    let start = Math.max(1, page - Math.floor(maxVisible / 2))
   410→    const end = Math.min(totalPages, start + maxVisible - 1)
   411→    if (end - start + 1 < maxVisible) start = Math.max(1, end - maxVisible + 1)
   412→    for (let i = start; i <= end; i++) pages.push(i)
   413→    return pages
   414→  }
   415→
   416→  const hasActiveFilters = search || stage || contactId || companyId || ownerId
   417→  const clearFilters = () => {
   418→    setSearch(''); setStage(''); setContactId(''); setCompanyId(''); setOwnerId(''); setSort('createdAt-desc')
   419→  }
   420→
   421→  // Group deals by stage for Kanban
   422→  const dealsByStage = DEAL_STAGES.map((s) => ({
   423→    ...s,
   424→    deals: allDeals.filter((d) => d.stage === s.key),
   425→  }))
   426→
   427→  return (
   428→    <div className='space-y-6'>
   429→      {/* Header */}
   430→      <div className='flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between'>
   431→        <div>
   432→          <h1 className='text-2xl font-bold tracking-tight'>Deals</h1>
   433→          <p className='text-muted-foreground mt-1'>Manage your sales pipeline</p>
   434→        </div>
   435→        <div className='flex items-center gap-2'>
   436→          <Button
   437→            variant={viewMode === 'table' ? 'default' : 'outline'}
   438→            size='icon'
   439→            onClick={() => setViewMode('table')}
   440→            aria-label='Table view'
   441→          >
   442→            <List className='size-4' />
   443→          </Button>
   444→          <Button
   445→            variant={viewMode === 'kanban' ? 'default' : 'outline'}
   446→            size='icon'
   447→            onClick={() => setViewMode('kanban')}
   448→            aria-label='Kanban view'
   449→          >
   450→            <LayoutGrid className='size-4' />
   451→          </Button>
   452→          <Button className='min-w-[130px]' onClick={() => setFormOpen(true)}>
   453→            <Plus className='size-4' /> Create Deal
   454→          </Button>
   455→        </div>
   456→      </div>
   457→
   458→      {/* Search and Filters */}
   459→      <div className='flex flex-col gap-3'>
   460→        <div className='relative'>
   461→          <Search className='absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground' />
   462→          <Input
   463→            placeholder='Search deals...'
   464→            value={search}
   465→            onChange={(e) => setSearch(e.target.value)}
   466→            className='pl-9'
   467→            aria-label='Search deals'
   468→          />
   469→        </div>
   470→

   477→          <Select value={stage} onValueChange={(v) => setStage(v === '_all' ? '' : v)}>
