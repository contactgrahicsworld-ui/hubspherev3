    12→const results = { passed: 0, failed: 0, errors: [], warnings: [] };
    13→
    14→function log(category, test, status, detail = '') {
    15→  const icon = status === 'PASS' ? '✅' : status === 'WARN' ? '⚠️' : '❌';
    16→  const msg = `${icon} [${category}] ${test}${detail ? ': ' + detail : ''}`;
    17→  console.log(msg);
    18→  if (status === 'PASS') results.passed++;
    19→  else if (status === 'WARN') results.warnings.push(`[${category}] ${test}: ${detail}`);
    20→  else { results.failed++; results.errors.push(`[${category}] ${test}: ${detail}`); }
    21→}

    23→async function api(path, options = {}) {
    24→  const url = `${BASE}${path}`;
    25→  const headers = { 'Content-Type': 'application/json', ...options.headers };
    26→  if (accessToken && !options.noAuth) headers['Authorization'] = `Bearer ${accessToken}`;
    27→  try {
    28→    const res = await fetch(url, { ...options, headers });
    29→    const text = await res.text();
    30→    let data;
    31→    try { data = JSON.parse(text); } catch { data = text; }
    32→    return { status: res.status, data, ok: res.ok, headers: res.headers };
    33→  } catch (e) {
    34→    return { status: 0, data: null, ok: false, error: e.message };
    35→  }
    36→}

   262→  let res = await api('/api/v1/auth/refresh', {
   263→    method: 'POST',
   264→    body: JSON.stringify({ refreshToken }),
   265→    noAuth: true
   266→  });
   267→  log('Auth-Refresh', 'Refresh token works', res.ok ? 'PASS' : 'FAIL', `status=${res.status}`);
   268→
   269→  if (res.ok && res.data?.data) {
   270→    const newAccess = res.data.data.accessToken;
   271→    log('Auth-Refresh', 'Got new access token', newAccess ? 'PASS' : 'FAIL');
   272→    // New refresh token is in Set-Cookie header (httpOnly)
   273→    const setCookie = res.headers.get('set-cookie') || '';
   274→    const hasRefreshCookie = setCookie.includes('hs-refresh-token');
   275→    log('Auth-Refresh', 'New refresh token set in cookie', hasRefreshCookie ? 'PASS' : 'WARN', `cookie found=${hasRefreshCookie}`);
   276→
   277→    // Old refresh token should be revoked
   278→    const oldRes = await api('/api/v1/auth/refresh', {
   279→      method: 'POST',
   280→      body: JSON.stringify({ refreshToken }),
   281→      noAuth: true
   282→    });
   283→    log('Auth-Refresh', 'Old refresh token revoked', oldRes.status === 401 ? 'PASS' : 'FAIL', `status=${oldRes.status}`);
   284→
   285→    if (newAccess) accessToken = newAccess;
   286→    // Extract new refresh token from cookie for subsequent tests
   287→    const match = setCookie.match(/hs-refresh-token=([^;]+)/);
   288→    if (match) refreshToken = match[1];
   289→  }
   290→
   291→  res = await api('/api/v1/auth/refresh', {
   292→    method: 'POST',
   293→    body: JSON.stringify({ refreshToken: 'invalid-token-xyz' }),
   294→    noAuth: true
   295→  });
   296→  log('Auth-Refresh', 'Invalid refresh token rejected', res.status === 401 ? 'PASS' : 'FAIL', `status=${res.status}`);
   297→}
