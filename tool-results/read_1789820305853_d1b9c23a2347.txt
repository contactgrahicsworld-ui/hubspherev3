    96→function formatDateTime(dateStr: string | null): string {
    97→  if (!dateStr) return '-';
    98→  try {
    99→    return new Date(dateStr).toLocaleString('en-US', {
   100→      month: 'short',
   101→      day: 'numeric',
   102→      year: 'numeric',
   103→      hour: 'numeric',
   104→      minute: '2-digit',
   105→    });
   106→  } catch {
   107→    return '-';
   108→  }
   109→}
   110→
   111→function formatDuration(seconds: number | null): string {
   112→  if (seconds == null) return '-';
   113→  const m = Math.floor(seconds / 60);
   114→  const s = seconds % 60;
   115→  return m > 0 ? `${m}m ${s}s` : `${s}s`;
   116→}
