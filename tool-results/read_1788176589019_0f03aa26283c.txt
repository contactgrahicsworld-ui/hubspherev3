   129→function getLeadName(lead: Lead): string {
   130→  return [lead.firstName, lead.lastName].filter(Boolean).join(' ') || '-'
   131→}
   132→
   133→function formatDateTime(dateStr: string) {
   134→  if (!dateStr) return '-'
   135→  try {
   136→    return new Date(dateStr).toLocaleString('en-US', {
   137→      month: 'short',
   138→      day: 'numeric',
   139→      year: 'numeric',
   140→      hour: 'numeric',
   141→      minute: '2-digit',
   142→    })
   143→  } catch {
   144→    return '-'
   145→  }
   146→}
   147→
   148→function formatDuration(seconds: number | null): string {
   149→  if (seconds == null) return '-'
   150→  const m = Math.floor(seconds / 60)
   151→  const s = seconds % 60
   152→  return m > 0 ? `${m}m ${s}s` : `${s}s`
   153→}
   154→
   155→function decodeToken(token: string): { userId: string; email: string } | null {
   156→  try {
   157→    const parts = token.split('.')
   158→    if (parts.length < 2) return null
   159→    const payload = JSON.parse(atob(parts[1]))
   160→    return {
   161→      userId: payload.sub || payload.userId || payload.id || '',
   162→      email: payload.email || '',
   163→    }
   164→  } catch {
   165→    return null
   166→  }
   167→}
