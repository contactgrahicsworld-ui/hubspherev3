    96→    96→function formatDateTime(dateStr: string | null): string {
    97→    97→  if (!dateStr) return '-';
    98→    98→  try {
    99→    99→    return new Date(dateStr).toLocaleString('en-US', {
   100→   100→      month: 'short',
   101→   101→      day: 'numeric',
   102→   102→      year: 'numeric',
   103→   103→      hour: 'numeric',
   104→   104→      minute: '2-digit',
   105→   105→    });
   106→   106→  } catch {
   107→   107→    return '-';
   108→   108→  }
   109→   109→}
   110→   110→
   111→   111→function formatDuration(seconds: number | null): string {
   112→   112→  if (seconds == null) return '-';
   113→   113→  const m = Math.floor(seconds / 60);
   114→   114→  const s = seconds % 60;
   115→   115→  return m > 0 ? `${m}m ${s}s` : `${s}s`;
   116→   116→}
